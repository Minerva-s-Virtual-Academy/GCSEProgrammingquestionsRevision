# Q02cii - Complete Python Solution

HEXDIV = 16
userInput = 0

def getLetter(hexDigit):
    if hexDigit == 10:
        return "A"
    elif hexDigit == 11:
        return "B"
    elif hexDigit == 12:
        return "C"
    elif hexDigit == 13:
        return "D"
    elif hexDigit == 14:
        return "E"
    elif hexDigit == 15:
        return "F"
    else:
        return str(hexDigit)

def getHex(denaryValue):
    hexFirst = denaryValue // HEXDIV
    hexSecond = denaryValue % HEXDIV
    return getLetter(hexFirst) + getLetter(hexSecond)

# Complete conversion to integer
userInput = int(input("please enter a number between 0 and 255 \n"))

# Complete test for range 0 to 255
if userInput >= 0 and userInput <= 255:
    # Call getHex with userInput argument
    print(getHex(userInput))
else:
    print("number out of range")

# Q04c - Model Python Solution

import random

# Program constants
SIDES = 6
THROWS = 3

# Initialise variables
roll = 0
subtotal = 0
booster = 1
finalScore = 0

# Roll dice 3 times
for i in range(THROWS):
    # Display number rolled
    roll = random.randint(1, SIDES)
    print("Rolled:", roll)
    
    # Update subtotal
    subtotal = subtotal + roll
    
    # Check for roll of 3 or 6 and increment booster if needed
    if roll == 3 or roll == 6:
        booster = booster + 1

# Calculate and output final score
finalScore = subtotal * booster
print("Score", finalScore)
